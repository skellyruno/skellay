cmake_minimum_required(VERSION 3.22)
project(SkellayNative VERSION 0.9.0 LANGUAGES C CXX)
add_library(skellay_native STATIC native/Delay.cpp native/FruityDelayBank.cpp native/SkellayEngine.cpp)
target_compile_features(skellay_native PUBLIC cxx_std_17)
target_include_directories(skellay_native PUBLIC native)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(skellay_native PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
elseif(MSVC)
 target_compile_options(skellay_native PRIVATE /W4 /fp:strict /utf-8)
endif()
enable_testing()
add_executable(classic_modern tests/classic_modern.cpp native/bridge.cpp)
target_link_libraries(classic_modern PRIVATE skellay_native)
add_test(NAME classic_modern COMMAND classic_modern)
add_executable(matrix tests/matrix.cpp)
target_link_libraries(matrix PRIVATE skellay_native)
add_test(NAME matrix COMMAND matrix)
target_sources(skellay_native PRIVATE native/PresetImport.cpp)
add_executable(preset_import tests/preset_import.cpp)
target_link_libraries(preset_import PRIVATE skellay_native)
add_test(NAME preset_import COMMAND preset_import)
add_executable(skellay-preset tools/preset_cli.cpp)
target_link_libraries(skellay-preset PRIVATE skellay_native)
find_package(Threads REQUIRED)
add_executable(engine_exchange tests/exchange.cpp)
target_link_libraries(engine_exchange PRIVATE skellay_native Threads::Threads)
add_test(NAME engine_exchange COMMAND engine_exchange)
option(SKELLAY_BUILD_VST3 "Build experimental JUCE VST3" ON)
if(SKELLAY_BUILD_VST3)
  include(FetchContent)
  if(DEFINED JUCE_PATH AND EXISTS "${JUCE_PATH}/CMakeLists.txt")
    add_subdirectory("${JUCE_PATH}" JUCE)
  else()
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.12 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
  endif()
  set_target_properties(skellay_native PROPERTIES POSITION_INDEPENDENT_CODE ON)
  juce_add_plugin(Skellay COMPANY_NAME "Skell Audio" BUNDLE_ID "audio.skell.skellay" PLUGIN_MANUFACTURER_CODE SkAu PLUGIN_CODE Slya FORMATS VST3 PRODUCT_NAME "Skellay" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(Skellay PRIVATE plugin/Skellay.cpp)
  target_compile_features(Skellay PRIVATE cxx_std_17)
  target_compile_definitions(Skellay PUBLIC JUCE_VST3_CAN_REPLACE_VST2=0 PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(Skellay PRIVATE skellay_native Threads::Threads juce::juce_audio_utils juce::juce_gui_extra)
  juce_add_console_app(SkellayEditorCheck PRODUCT_NAME "Skellay Editor Check")
  target_sources(SkellayEditorCheck PRIVATE ui/smoke.cpp)
  target_compile_features(SkellayEditorCheck PRIVATE cxx_std_17)
  target_compile_definitions(SkellayEditorCheck PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(SkellayEditorCheck PRIVATE juce::juce_gui_extra)
endif()
